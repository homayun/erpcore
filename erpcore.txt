function Export-Part($name, $includePatterns) {
    $out = "D:\SpringWorkSpace\erpcore-$name.txt"
    $sb = New-Object System.Text.StringBuilder
    [void]$sb.AppendLine("=== ERPCORE PROJECT EXPORT - PART: $name ===")
    [void]$sb.AppendLine("Generated: $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')")
    foreach ($pat in $includePatterns) {
        $files = Get-ChildItem -Path $pat -Recurse -File -ErrorAction SilentlyContinue | Where-Object {
            @('.java','.xml','.yml','.yaml','.properties','.sql','.html','.css','.js','.md','.json') -contains $_.Extension.ToLower() -and
            $_.Name -notlike '*.bak'
        }
        foreach ($f in $files) {
            $rel = $f.FullName.Substring("D:\SpringWorkSpace\erpcore".Length + 1)
            [void]$sb.AppendLine("`n" + ("=" * 90))
            [void]$sb.AppendLine("FILE: $rel")
            [void]$sb.AppendLine("=" * 90)
            try { [void]$sb.AppendLine((Get-Content $f.FullName -Raw -Encoding UTF8)) }
            catch { [void]$sb.AppendLine("<<ERROR: $_>>") }
        }
    }
    $utf8NoBom = New-Object System.Text.UTF8Encoding $false
    [System.IO.File]::WriteAllText($out, $sb.ToString(), $utf8NoBom)
    $kb = [math]::Round((Get-Item $out).Length / 1KB, 1)
    Write-Host "✅ $name → $out ($kb KB)" -ForegroundColor Green
}

Export-Part "01-security" @("src\main\java\com\erp\core\security")
Export-Part "02-admin"    @("src\main\java\com\erp\core\admin")
Export-Part "03-web"      @("src\main\resources\templates", "src\main\resources\static")
Export-Part "04-config"   @("src\main\resources\db", "examples\quickstart-app\src\main\resources", "pom.xml")
